## 第二步：实现核心函数

### 计算平均分

```python
def calc_average(scores):
    if len(scores) == 0:
        return 0
    return sum(scores) / len(scores)
```

### 判断及格

```python
def is_pass(score):
    return score >= 60
```

### 添加学生

```python
def add_student(students, name):
    for s in students:
        if s["name"] == name:
            print(f"学生 {name} 已存在！")
            return
    students.append({"name": name, "scores": []})
    print(f"✅ 已添加学生：{name}")
```

### 录入成绩

```python
def add_score(students, name, score):
    for s in students:
        if s["name"] == name:
            s["scores"].append(score)
            print(f"✅ {name} 成绩 {score} 已录入")
            return
    print(f"❌ 未找到学生：{name}")
```

### 查询学生

```python
def query_student(students, name):
    for s in students:
        if s["name"] == name:
            avg = calc_average(s["scores"])
            status = "及格" if avg >= 60 else "不及格"
            print(f"姓名：{s['name']}")
            print(f"成绩：{s['scores']}")
            print(f"平均分：{avg:.1f} ({status})")
            return
    print(f"❌ 未找到学生：{name}")
```

### 排名显示

```python
def show_ranking(students):
    # 计算每个学生的平均分并排序
    ranked = []
    for s in students:
        avg = calc_average(s["scores"])
        ranked.append({"name": s["name"], "average": avg})

    # 降序排序
    ranked.sort(key=lambda x: x["average"], reverse=True)

    print("\n📊 成绩排名：")
    for i, r in enumerate(ranked, 1):
        print(f"  {i}. {r['name']}: {r['average']:.1f}")
```
